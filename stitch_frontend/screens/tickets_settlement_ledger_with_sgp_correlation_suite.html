import numpy as np
import scipy.stats as stats

# =====================================================================
# LIVEINTEL TERMINAL // CORRELATED SGP & MIDDLE HEDGE BACKTEST ENGINE
# KELLY CRITERION SIZING PARAMETERS & COPULA JOINT DISTRIBUTION MODEL
# =====================================================================
KELLY_FRACTION = 0.5           # Half-Kelly multiplier (0.125 to 1.0)
BANKROLL_BASE = 10480.00       # Synced ledger starting capital
MAX_STAKE_PCT = 0.05           # Cap single leg at 5% max exposure
CORRIDOR_PROB_THRESHOLD = 0.15 # Minimum 15% joint middle probability
HEDGE_OPTIMIZATION_MODE = 'MAX_GROWTH'  # Options: MAX_GROWTH, ZERO_LOSS, EQUAL_PAYOFF

def calculate_kelly_hedge_allocation(prob_a, odds_a, prob_b, odds_b, correlation_rho, kelly_fraction=KELLY_FRACTION):
    """Model synthetic bivariate normal joint distribution, dual-leg Kelly fraction calculation,
    guaranteed floor risk, and double-hit sweet spot payoff."""
    dec_a = (odds_a / 100.0 + 1.0) if odds_a > 0 else (100.0 / abs(odds_a) + 1.0)
    dec_b = (odds_b / 100.0 + 1.0) if odds_b > 0 else (100.0 / abs(odds_b) + 1.0)
    b_a = dec_a - 1.0
    b_b = dec_b - 1.0
    
    # Bivariate Gaussian Copula joint middle / corridor probability
    z_a = stats.norm.ppf(prob_a)
    z_b = stats.norm.ppf(prob_b)
    joint_middle_prob = stats.multivariate_normal.cdf([z_a, z_b], mean=[0, 0], cov=[[1.0, correlation_rho], [correlation_rho, 1.0]])
    
    # Optimal dual-leg Kelly staking with correlation attenuation
    kelly_raw_a = (prob_a * b_a - (1.0 - prob_a)) / b_a
    kelly_raw_b = (prob_b * b_b - (1.0 - prob_b)) / b_b
    corr_shrinkage = 1.0 - (abs(correlation_rho) * 0.35)
    
    stake_pct_a = min(MAX_STAKE_PCT, max(0.0, kelly_raw_a * kelly_fraction * corr_shrinkage))
    stake_pct_b = min(MAX_STAKE_PCT, max(0.0, kelly_raw_b * kelly_fraction * corr_shrinkage))
    stake_a = round(BANKROLL_BASE * stake_pct_a, 2)
    stake_b = round(BANKROLL_BASE * stake_pct_b, 2)
    
    # Settlement payoff state spaces
    win_a_only = (stake_a * b_a) - stake_b
    win_b_only = (stake_b * b_b) - stake_a
    double_hit_win = (stake_a * b_a) + (stake_b * b_b)
    both_lose = -(stake_a + stake_b)
    zero_loss_floor = min(win_a_only, win_b_only)
    
    return {
        'stake_leg_a': stake_a,
        'stake_leg_b': stake_b,
        'joint_corridor_prob': round(joint_middle_prob, 4),
        'guaranteed_floor': round(zero_loss_floor, 2),
        'double_dip_payoff': round(double_hit_win, 2),
        'opt_mode': HEDGE_OPTIMIZATION_MODE
    }

def simulate_monte_carlo_drawdown(n_sims=10000, n_slates=100):
    np.random.seed(42)
    alloc = calculate_kelly_hedge_allocation(0.535, 108, 0.512, -102, 0.567)
    daily_returns = np.random.normal(loc=0.0182, scale=0.046, size=(n_sims, n_slates))
    cumulative_pnl = np.cumprod(1.0 + daily_returns * KELLY_FRACTION, axis=1)
    running_max = np.maximum.accumulate(cumulative_pnl, axis=1)
    drawdowns = (cumulative_pnl - running_max) / running_max
    max_dd = np.abs(np.min(drawdowns, axis=1))
    return alloc, np.percentile(max_dd, 95), np.percentile(max_dd, 99)

# =====================================================================
# LIVEINTEL TERMINAL // R QUANTITATIVE BACKTEST SCRIPT
# KELLY CRITERION SIZING & JOINT MIDDLE HEDGE MODEL
# =====================================================================
library(mvtnorm)

# Synced Ledger Parameters
kelly_fraction <- 0.5              # Half-Kelly multiplier
bankroll_base <- 10480.00          # Synced starting ledger bankroll
max_stake_pct <- 0.05              # 5% max single leg risk cap
corridor_prob_threshold <- 0.15    # 15% joint sweet-spot threshold
hedge_optimization_mode <- "MAX_GROWTH"

calc_kelly_middle_hedge <- function(prob_a = 0.535, odds_a = 108,
                                     prob_b = 0.512, odds_b = -102,
                                     correlation_rho = 0.567,
                                     k_frac = kelly_fraction,
                                     b_base = bankroll_base) {
  dec_a <- if (odds_a > 0) (odds_a / 100 + 1) else (100 / abs(odds_a) + 1)
  dec_b <- if (odds_b > 0) (odds_b / 100 + 1) else (100 / abs(odds_b) + 1)
  b_a <- dec_a - 1
  b_b <- dec_b - 1
  
  # Bivariate Joint Normal Corridor Probability
  sigma <- matrix(c(1.0, correlation_rho, correlation_rho, 1.0), nrow = 2)
  joint_prob <- pmvnorm(lower = c(-Inf, -Inf), upper = qnorm(c(prob_a, prob_b)), sigma = sigma)[1]
  
  # Correlated Kelly Stake Calculations
  k_raw_a <- (prob_a * b_a - (1 - prob_a)) / b_a
  k_raw_b <- (prob_b * b_b - (1 - prob_b)) / b_b
  shrinkage <- 1.0 - (abs(correlation_rho) * 0.35)
  
  stake_a <- round(b_base * min(max_stake_pct, max(0, k_raw_a * k_frac * shrinkage)), 2)
  stake_b <- round(b_base * min(max_stake_pct, max(0, k_raw_b * k_frac * shrinkage)), 2)
  
  # Sweet spot payoff calculation
  floor_lock <- min((stake_a * b_a) - stake_b, (stake_b * b_b) - stake_a)
  double_dip <- (stake_a * b_a) + (stake_b * b_b)
  
  list(
    stake_leg_a = stake_a,
    stake_leg_b = stake_b,
    corridor_prob = round(joint_prob, 4),
    guaranteed_floor = floor_lock,
    sweet_spot_win = double_dip,
    optimization_mode = hedge_optimization_mode
  )
}

# Monte Carlo Simulation (N = 10,000 runs)
run_simulations <- function(n_runs = 10000, n_slates = 100) {
  set.seed(42)
  hedge <- calc_kelly_middle_hedge()
  returns <- matrix(rnorm(n_runs * n_slates, mean = 0.0182, sd = 0.046), nrow = n_runs)
  pnl_paths <- t(apply(1 + returns * kelly_fraction, 1, cumprod))
  run_max <- t(apply(pnl_paths, 1, cummax))
  dd <- (pnl_paths - run_max) / run_max
  max_dd <- apply(abs(dd), 1, max)
  
  cat("=== LIVEINTEL R BACKTEST RESULTS ===\n")
  cat(sprintf("Leg A Stake: $%.2f | Leg B Hedge: $%.2f\n", hedge$stake_leg_a, hedge$stake_leg_b))
  cat(sprintf("Guaranteed Floor Lock: +$%.2f | Double-Dip Win: +$%.2f\n", hedge$guaranteed_floor, hedge$sweet_spot_win))
  cat(sprintf("95%% VaR: -%.2f%% | 99%% Max DD: -%.2f%%\n", quantile(max_dd, 0.95) * 100, quantile(max_dd, 0.99) * 100))
}

run_simulations()
